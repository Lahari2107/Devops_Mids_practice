Step 3: Create the Selenium test script
Create a file named test.js.

This script connects to the Chrome browser running inside the Selenium Docker container.
// Import Selenium WebDriver and assert module
const { Builder, By } = require('selenium-webdriver');
const assert = require('assert');

async function waitForSelenium() {

    const url = process.env.SELENIUM_REMOTE_URL;

    // Wait until Selenium server is ready
    for (let i = 0; i < 30; i++) {
        try {
            const response = await fetch(url + '/status');
            const data = await response.json();

            if (data.value && data.value.ready) {
                console.log("Selenium server is ready!");
                return;
            }
        } catch (error) {
            // Try again if Selenium is not ready
        }

        await new Promise(resolve => setTimeout(resolve, 2000));
    }

    throw new Error("Selenium server did not become ready.");
}

async function runTest() {

    let driver;

    try {

        // Wait for Selenium server
        await waitForSelenium();

        // Connect to remote Chrome browser
        driver = await new Builder()
            .usingServer(process.env.SELENIUM_REMOTE_URL)
            .forBrowser('chrome')
            .build();

        // Open the registration application
        await driver.get('http://app');

        // Enter registration details
        await driver.findElement(By.id('name'))
            .sendKeys('Lahari');

        await driver.findElement(By.id('email'))
            .sendKeys('lahari@example.com');

        // Click the Register button
        await driver.findElement(By.id('submitBtn')).click();

        // Retrieve the success message
        let message = await driver.findElement(By.id('message'));

        let actualMessage = await message.getText();

        console.log("Actual Message: " + actualMessage);

        // Validate the expected result
        assert.strictEqual(
            actualMessage,
            'Registration Successful!'
        );

        console.log("Test Passed!");
        console.log("Selenium test executed successfully inside Docker.");

    } catch (error) {

        console.error("Test Failed:", error);
        process.exitCode = 1;

    } finally {

        // Close browser session
        if (driver) {
            await driver.quit();
        }

    }
}

// Execute the test
runTest();

Step 5: Create the Dockerfile
Create a file named Dockerfile (without any extension).
# Use Node.js as the base image
FROM node:20-bookworm-slim

# Set working directory
WORKDIR /app

# Copy package.json
COPY package.json ./

# Install project dependencies
RUN npm install

# Copy Selenium test script
COPY test.js ./

# Execute the Selenium test
CMD ["npm", "test"]


This Dockerfile packages Node.js, Selenium WebDriver and the testing script into a Docker image.
Step 6: Create the Docker Compose file
Create a file named docker-compose.yml.
services:

  # Registration application container
  app:
    image: nginx:alpine
    volumes:
      - ./index.html:/usr/share/nginx/html/index.html:ro

  # Selenium Chrome browser container
  selenium:
    image: selenium/standalone-chrome:latest
    shm_size: 2gb

  # Selenium test execution container
  test:
    build: .
    depends_on:
      - app
      - selenium
    environment:
      SELENIUM_REMOTE_URL: http://selenium:4444


How the containers work:
- app: Runs the registration webpage using Nginx.
- selenium: Provides the Chrome browser and Selenium server.
- test: Runs your JavaScript Selenium automation script.
Docker Compose connects these containers through a shared network.

2. Build the Selenium test image.
3. Start the registration application.
4. Start the Selenium Chrome browser.
5. Execute the JavaScript test inside the test container.

Step 2: Update the Selenium test script
Open test.js and replace its entire contents with the following code.

This script performs three automated test cases:
1. Checks whether the registration page opens.
2. Enters user details into the form.
3. Submits the form and verifies the success message.
const { Builder, By, until } = require('selenium-webdriver');
const assert = require('assert');

async function waitForSelenium() {

    const url = process.env.SELENIUM_REMOTE_URL;

    // Wait until Selenium server is ready
    for (let i = 0; i < 30; i++) {
        try {
            const response = await fetch(url + '/status');
            const data = await response.json();

            if (data.value && data.value.ready) {
                console.log("Selenium server is ready!");
                return;
            }
        } catch (error) {
            // Retry if Selenium is not ready
        }

        await new Promise(resolve => setTimeout(resolve, 2000));
    }

    throw new Error("Selenium server is not available.");
}

async function runTests() {

    let driver;

    try {

        await waitForSelenium();

        // Connect to Chrome running inside Selenium container
        driver = await new Builder()
            .usingServer(process.env.SELENIUM_REMOTE_URL)
            .forBrowser('chrome')
            .build();

        // Open registration page
        await driver.get('http://app');

        // TEST CASE 1: Verify page title
        let title = await driver.getTitle();

        assert.strictEqual(title, 'Docker Selenium Testing');

        console.log("Test Case 1 Passed: Page title verified.");

        // TEST CASE 2: Enter registration details
        await driver.findElement(By.id('name'))
            .sendKeys('Lahari');

        await driver.findElement(By.id('email'))
            .sendKeys('lahari@example.com');

        let enteredName = await driver.findElement(By.id('name')).getAttribute('value');

        assert.strictEqual(enteredName, 'Lahari');

        console.log("Test Case 2 Passed: User input verified.");

        // TEST CASE 3: Submit the registration form
        await driver.findElement(By.id('submitBtn')).click();

        let message = await driver.wait(
            until.elementLocated(By.id('message')),
            5000
        );

        let actualMessage = await message.getText();

        assert.strictEqual(
            actualMessage,
            'Registration Successful!'
        );

        console.log("Test Case 3 Passed: Success message verified.");

        console.log("All Selenium test cases passed!");

    } catch (error) {

        console.error("Test execution failed:", error);
        process.exitCode = 1;

    } finally {

        // Close browser session
        if (driver) {
            await driver.quit();
        }

    }
}

runTests();
