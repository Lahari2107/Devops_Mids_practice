tep 3: Create a Java Selenium testing program
The lab manual uses Java Selenium with Microsoft Edge to open the calculator webpage and print its title. 


In your NetBeans project, create a Java class named TestCalculator and paste:


import org.openqa.selenium.WebDriver;
import org.openqa.selenium.edge.EdgeDriver;

public class TestCalculator {

    public static void main(String[] args) {

        WebDriver driver = new EdgeDriver();

        try {
            // Open the JavaScript calculator
            driver.get("file:///C:/DevOpsLab/Set6/JavaScriptCalculator.html");

            // Print the page title
            System.out.println("Title: " + driver.getTitle());

            // Keep the browser open for 60 seconds
            Thread.sleep(60000);

        } catch (Exception e) {
            System.out.println("Error: " + e.getMessage());

        } finally {
            // Close the browser
            driver.quit();
        }
    }
}

Step 5: Create the Selenium automation script
Create another file named formClick.js in the same folder.

Paste the complete code:
// Import Selenium WebDriver
const { Builder, By, until } = require('selenium-webdriver');

async function testFormClick() {

    // Launch Google Chrome browser
    let driver = await new Builder().forBrowser('chrome').build();

    try {

        // Open the HTML form
        await driver.get('file://' + __dirname + '/index.html');

        // Locate the button using its ID
        let submitButton = await driver.findElement(By.id('submitBtn'));

        // Wait until the button is clickable
        await driver.wait(until.elementIsEnabled(submitButton), 5000);

        // Click the button
        await submitButton.click();

        console.log("Submit button clicked successfully!");

        // Verify the success message
        let message = await driver.findElement(By.id('message'));

        let actualMessage = await message.getText();

        console.log("Message: " + actualMessage);

        if (actualMessage === "Form submitted successfully!") {
            console.log("Test Passed!");
        } else {
            console.log("Test Failed!");
        }

    } catch (error) {

        console.log("Error:", error);

    } finally {

        // Close the browser
        await driver.quit();

    }
}

// Execute the test function
testFormClick();
