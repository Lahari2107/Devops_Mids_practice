notepad Dockerfile

# Use the official Nginx image from Docker Hub
FROM nginx:latest

# Copy the HTML file into the Nginx container
COPY EventRegistrationForm.html /usr/share/nginx/html/index.html

# Expose port 8082
EXPOSE 8082

# Start Nginx in the foreground
CMD ["nginx", "-g", "daemon off;"]

# 1. Navigate to the project directory
cd Docker-Event-Registration

# 2. Build the custom Docker image
docker build -t my-web-app .

# 3. Verify the image
docker images

# 4. Run the container
docker run -d --name event-registration -p 8082:80 my-web-app

# 5. Check running containers
docker ps

# 6. Check container logs
docker logs event-registration

# 7. Stop the container
docker stop event-registration

# 8. Remove the container
docker rm event-registration

# 9. Verify removal
docker ps -a
