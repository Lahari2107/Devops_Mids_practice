Step 3: Selenium test script – test.js
Replace the entire contents of test.js with:
const { Builder, By, until } = require('selenium-webdriver');
const assert = require('assert');

// Wait until both the application and Selenium server are ready
async function waitForServices() {

    const appURL = 'http://app';
    const seleniumURL = process.env.SELENIUM_REMOTE_URL;

    for (let i = 0; i < 30; i++) {

        try {
            const appResponse = await fetch(appURL);
            const seleniumResponse = await fetch(
                seleniumURL + '/status'
            );

            const seleniumData = await seleniumResponse.json();

            if (appResponse.ok && seleniumData.value.ready) {
                console.log("Application and Selenium are ready!");
                return;
            }

        } catch (error) {
            // Retry until both services are ready
        }

        await new Promise(resolve => setTimeout(resolve, 2000));
    }

    throw new Error("Application or Selenium service is unavailable.");
}

async function runSeleniumTests() {

    let driver;

    try {

        // Wait for the required containers
        await waitForServices();

        // Connect to Chrome inside the Selenium container
        driver = await new Builder()
            .usingServer(process.env.SELENIUM_REMOTE_URL)
            .forBrowser('chrome')
            .build();

        // Open the containerized application
        await driver.get('http://app');

        // TEST 1: Verify application title
        let title = await driver.getTitle();

        assert.strictEqual(title, 'Set 9 Selenium Testing');

        console.log("Test 1 Passed: Application opened successfully.");

        // TEST 2: Enter registration details
        await driver.findElement(By.id('name'))
            .sendKeys('Lahari');

        await driver.findElement(By.id('email'))
            .sendKeys('lahari@example.com');

        let enteredName = await driver.findElement(By.id('name'))
            .getAttribute('value');

        assert.strictEqual(enteredName, 'Lahari');

        console.log("Test 2 Passed: User details entered successfully.");

        // TEST 3: Submit the registration form
        await driver.findElement(By.id('submitBtn')).click();

        let message = await driver.wait(
            until.elementLocated(By.id('message')),
            5000
        );

        let actualMessage = await message.getText();

        assert.strictEqual(
            actualMessage,
            'Registration Successful!'
        );

        console.log("Test 3 Passed: Success message verified.");

        console.log("All Selenium tests passed successfully!");

    } catch (error) {

        console.error("Test execution failed:", error);
        process.exitCode = 1;

    } finally {

        // Close the browser session
        if (driver) {
            await driver.quit();
        }

    }
}

// Execute the test script
runSeleniumTests();

Step 6: Create docker-compose.yml
Create or replace docker-compose.yml:
services:

  # Container 1: Registration application
  app:
    image: nginx:alpine
    volumes:
      - ./index.html:/usr/share/nginx/html/index.html:ro

  # Container 2: Selenium Chrome browser
  selenium:
    image: selenium/standalone-chrome:latest
    shm_size: 2gb

  # Container 3: Selenium test execution
  test:
    build: .
    depends_on:
      - app
      - selenium
    environment:
      SELENIUM_REMOTE_URL: http://selenium:4444

# Lab Implementation Report – Selenium Containerization and Automated Testing

### Aim
To containerize a Selenium automation testing environment using Docker and execute automated test cases against a web application.

### Tools and Technologies Used
- JavaScript
- Node.js
- Selenium WebDriver
- Docker
- Docker Compose
- Nginx
- Chrome Browser

### Implementation Steps

1. Created a registration form using HTML with name and email input fields and a Register button.
2. Developed a Selenium automation script using JavaScript to interact with the registration form.
3. Created a Dockerfile to package Node.js, Selenium dependencies and the test script into a Docker image.
4. Configured Docker Compose to run three containers:
   - Application container using Nginx.
   - Selenium container using Standalone Chrome.
   - Test container to execute the JavaScript automation script.
5. Built the Docker image using `docker compose build`.
6. Started the containers and executed the Selenium tests using `docker compose up`.
7. Verified the application title, user input and registration success message.
8. Stopped and removed the containers using `docker compose down`.

### Test Cases Performed

| Test Case | Expected Result |
|---|---|
| Verify application title | Correct page title displayed |
| Enter registration details | User details entered successfully |
| Submit registration form | Registration success message displayed |

### Result
All three Selenium test cases were executed successfully inside Docker containers. The application and browser testing environment operated independently of the host system.

### Conclusion
Selenium containerization using Docker provides an isolated and consistent environment for automated web application testing. Docker Compose simplifies the management and execution of the application, browser and testing containers.
