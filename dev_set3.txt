# deployment.yaml

apiVersion: apps/v1
kind: Deployment

metadata:
  name: event-registration

spec:
  replicas: 1

  selector:
    matchLabels:
      app: event-registration

  template:
    metadata:
      labels:
        app: event-registration

    spec:
      containers:
        - name: event-registration
          image: my-web-app:v1
          imagePullPolicy: IfNotPresent

          ports:
            - containerPort: 80


# service.yaml

apiVersion: v1
kind: Service

metadata:
  name: event-registration-service

spec:
  type: NodePort

  selector:
    app: event-registration

  ports:
    - port: 80
      targetPort: 80
      nodePort: 30080
----------------------------------------------------------------
setup.ps1

# ==========================================
# Docker and Kubernetes Container Setup
# ==========================================

# STEP 1: Check Docker installation
Write-Host "Checking Docker installation..."

docker --version

if ($LASTEXITCODE -ne 0) {
    Write-Host "Docker is not installed or running."
    exit 1
}

# STEP 2: Check Kubernetes connection
Write-Host "Checking Kubernetes cluster..."

kubectl get nodes

if ($LASTEXITCODE -ne 0) {
    Write-Host "Kubernetes cluster is not available."
    exit 1
}

# STEP 3: Build Docker image
Write-Host "Building Docker image..."

docker build -t my-web-app:v1 .

if ($LASTEXITCODE -ne 0) {
    Write-Host "Docker image build failed."
    exit 1
}

# STEP 4: Verify Docker image
Write-Host "Checking Docker image..."

docker images

# STEP 5: Apply Kubernetes Deployment
Write-Host "Creating Kubernetes Deployment..."

kubectl apply -f deployment.yaml

if ($LASTEXITCODE -ne 0) {
    Write-Host "Deployment failed."
    exit 1
}

# STEP 6: Apply Kubernetes Service
Write-Host "Creating Kubernetes Service..."

kubectl apply -f service.yaml

if ($LASTEXITCODE -ne 0) {
    Write-Host "Service creation failed."
    exit 1
}

# STEP 7: Check Deployment status
Write-Host "Checking Deployment..."

kubectl rollout status deployment/event-registration

# STEP 8: Display running Pods
Write-Host "Checking Kubernetes Pods..."

kubectl get pods

# STEP 9: Display Services
Write-Host "Checking Kubernetes Services..."

kubectl get services

# STEP 10: Display completion message
Write-Host "Container setup completed successfully!"

Write-Host "Application Service Port: 30080"
